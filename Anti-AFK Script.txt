local AllTextLabels = {}
local function RegisterText(lbl, isAccent)
    table.insert(AllTextLabels, {Label = lbl, IsAccent = isAccent})
    local base = isAccent and Theme.Accent or Theme.TextMain
    lbl.TextColor3 = Color3.new(
        math.clamp(base.R * Config.TextLightness, 0, 1),
        math.clamp(base.G * Config.TextLightness, 0, 1),
        math.clamp(base.B * Config.TextLightness, 0, 1)
    )
end

local function UpdateAllTextLightness()
    for _, item in ipairs(AllTextLabels) do
        if item.Label and item.Label.Parent then
            local base = item.IsAccent and Theme.Accent or Theme.TextMain
            item.Label.TextColor3 = Color3.new(
                math.clamp(base.R * Config.TextLightness, 0, 1),
                math.clamp(base.G * Config.TextLightness, 0, 1),
                math.clamp(base.B * Config.TextLightness, 0, 1)
            )
        end
    end
end

local function MakeDraggable(obj, handle)
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = obj.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

    elseif Config.Method == "Camera Nudge" then
        local cam = workspace.CurrentCamera
        if cam then
            cam.CFrame = cam.CFrame * CFrame.Angles(0, math.rad(2), 0)
            task.wait(0.05)
            cam.CFrame = cam.CFrame * CFrame.Angles(0, math.rad(-2), 0)
        end
    elseif Config.Method == "Micro Hop" then
        local char = Player.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
    end
end

    frameCount = frameCount + 1
    if tick() - lastFpsUpdate >= 0.5 then
        currentFps = math.floor(frameCount / (tick() - lastFpsUpdate))
        frameCount = 0
        lastFpsUpdate = tick()

        if currentFps < Session.LowestFps then
            Session.LowestFps = currentFps
        end

        local ping = 0
        pcall(function() ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)
        if ping > Session.PeakPing then
            Session.PeakPing = ping
        end

        PerfLabel.Text = string.format("FPS: %d  •  Ping: %dms  •  %s", currentFps, ping, Config.Keybind.Name)
        SWValue.Text = string.format("Peak Ping: %dms  •  Lowest FPS: %d", Session.PeakPing, Session.LowestFps)
    end

    if Config.Enabled then
        local elapsed = math.max(0, math.floor(tick() - Session.StartTime - Session.PausedAccumulated))
        local hrs = math.floor(elapsed / 3600)
        local mins = math.floor((elapsed % 3600) / 60)
        local secs = elapsed % 60
        local timeStr = string.format("%02d:%02d:%02d", hrs, mins, secs)
        TimerValue.Text = timeStr
        TWLabel.Text = string.format("🐾 %s", timeStr)
    end
end)
