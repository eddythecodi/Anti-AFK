local function ApplyDecoded(decoded)
    if type(decoded) ~= "table" then return end
    for k, v in pairs(decoded) do
        if k == "Keybind" then
            local ok, kc = pcall(function() return Enum.KeyCode[v] end)
            if ok and kc then Config.Keybind = kc end
        elseif k ~= "WebhookUrl" or v == "" or type(v) == "string" then
            if Config[k] ~= nil and type(Config[k]) == type(v) then
                Config[k] = v
            end
        end
    end
end

local function SaveConfig()
    if not writefile then return end
    pcall(function()
        local copy = table.clone(Config)
        copy.Keybind = Config.Keybind.Name
        writefile(FileName, HttpService:JSONEncode(copy))
    end)
end

local function LoadConfig()
    if isfile and readfile and isfile(FileName) then
        pcall(function()
            ApplyDecoded(HttpService:JSONDecode(readfile(FileName)))
        end)
    end
    Config.Enabled = true
    Config.TotalRuns = (Config.TotalRuns or 0) + 1
    SaveConfig()
end

local function Lit(c)

local function FormatDuration(s)
    s = math.max(0, math.floor(s))
    local h, m = math.floor(s / 3600), math.floor((s % 3600) / 60)
    if h > 0 then return string.format("%dh %dm", h, m) end
    if m > 0 then return string.format("%dm %ds", m, s % 60) end
    return string.format("%ds", s)
end

local function ActiveSeconds()
    local paused = Session.PausedAccumulated + (Session.LastPauseTime and (tick() - Session.LastPauseTime) or 0)
    return math.floor(tick() - Session.StartTime - paused)
end

local function PlaySFX(kind)
    if not Config.SoundEnabled or Config.SoundVolume <= 0.01 then return end
    local s = Instance.new("Sound")
    s.Volume = Config.SoundVolume
    if kind == "Click" then
        if Config.SoundProfile == "Bubble" then s.SoundId = "rbxassetid://9114221375"
        elseif Config.SoundProfile == "Cozy Chime" then s.SoundId = "rbxassetid://6026984224"
        else s.SoundId = "rbxassetid://6895079853" end
    elseif kind == "Toggle" then s.SoundId = "rbxassetid://452267918"
    elseif kind == "Notify" then s.SoundId = "rbxassetid://6026984224"
    elseif kind == "Alarm" then s.SoundId = "rbxassetid://5153724231"; s.Volume = 1
    end
    s.Parent = SoundService
    s:Play()
    s.Ended:Connect(function() s:Destroy() end)
    task.delay(6, function() if s and s.Parent then s:Destroy() end end)
end

local function IsValidWebhook(u)
    return type(u) == "string" and (
        u:match("^https://discord%.com/api/webhooks/") or
        u:match("^https://discordapp%.com/api/webhooks/") or
        u:match("^https://canary%.discord%.com/api/webhooks/") or
        u:match("^https://ptb%.discord%.com/api/webhooks/")
    ) ~= nil
end

local function SendWebhook(title, description)
    if not Config.WebhookEnabled or not IsValidWebhook(Config.WebhookUrl) then return end
    task.spawn(function()
        pcall(function()
            local req = (syn and syn.request) or http_request or request or (fluxus and fluxus.request)
            if not req then return end
            req({
                Url = Config.WebhookUrl,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = HttpService:JSONEncode({
                    embeds = {{
                        title = "🐾 Paw-Luxe: " .. title,
                        description = description,
                        color = 16738192,
                        footer = {text = "Player: " .. Player.Name},
                    }},
                }),
            })
        end)
    end)
end

local function MakeDraggable(obj, handle)
    local dragging, dragStart, startPos
    Track(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos = true, input.Position, obj.Position
            local c
            c = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    c:Disconnect()
                end
            end)
        end
    end))
    Track(UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end))
end

    local l = New("TextLabel", {
        BackgroundTransparency = 1, Text = text, Font = o.font or Enum.Font.GothamBold, TextSize = o.size or 12,
        TextXAlignment = o.xa or Enum.TextXAlignment.Left, TextYAlignment = o.ya or Enum.TextYAlignment.Center,
        Size = o.sz or UDim2.new(1, -24, 0, 22), Position = o.pos or UDim2.fromOffset(14, 10),
        TextWrapped = o.wrap or false, RichText = o.rich or false,
    }, parent)
    Txt(l, o.key)
    return l
end

local function Button(parent, text, size, pos, accent, ts)
    local b = New("TextButton", {
        Size = size, Position = pos, Text = text, TextSize = ts or 11, AutoButtonColor = false,
        Font = accent and Enum.Font.GothamBlack or Enum.Font.GothamBold,
    }, parent)
    Round(b, 8)
    if accent then
        Bind(b, "BackgroundColor3", "Accent")
        b.TextColor3 = DARK
    else
        Bind(b, "BackgroundColor3", "ButtonSolid")
        Txt(b)
    end
    return b
end

local function Input(parent, y, placeholder, text)
    local tb = New("TextBox", {
        Size = UDim2.new(1, -28, 0, 34), Position = UDim2.fromOffset(14, y), Text = text or "",
        PlaceholderText = placeholder, PlaceholderColor3 = Color3.fromRGB(150, 140, 170),
        Font = Enum.Font.GothamSemibold, TextSize = 11, ClearTextOnFocus = false,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, parent)
    Bind(tb, "BackgroundColor3", "ButtonSolid"); Txt(tb); Round(tb, 8)
    New("UIPadding", {PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8)}, tb)
    return tb
end

local function ToggleRow(parent, text, state, cb)
    local card = MakeCard(parent, 46)
    Label(card, text, {sz = UDim2.new(0.7, 0, 1, 0), pos = UDim2.fromOffset(14, 0)})
    local btn = New("TextButton", {
        Size = UDim2.fromOffset(64, 26), Position = UDim2.new(1, -76, 0.5, -13),
        Font = Enum.Font.GothamBlack, TextSize = 11, AutoButtonColor = false,
    }, card)
    Round(btn, 7)
    local function paint()
        btn.Text = state and "ON" or "OFF"
        StyleState(btn, state)
    end
    OnTheme(paint)
    Track(btn.MouseButton1Click:Connect(function()
        state = not state
        PlaySFX("Toggle")
        paint()
        cb(state)
        SaveConfig()
    end))
    return card, function(v) state = v; paint() end
end

local function StepperRow(parent, text, min, max, step, get, set, fmt)
    local card = MakeCard(parent, 46)
    Label(card, text, {sz = UDim2.new(0.5, 0, 1, 0), pos = UDim2.fromOffset(14, 0), size = 11})
    local minus = Button(card, "–", UDim2.fromOffset(28, 26), UDim2.new(1, -168, 0.5, -13))
    local val = Label(card, "", {sz = UDim2.fromOffset(92, 26), pos = UDim2.new(1, -136, 0.5, -13), xa = Enum.TextXAlignment.Center, key = "Accent", size = 11})
    local plus = Button(card, "+", UDim2.fromOffset(28, 26), UDim2.new(1, -40, 0.5, -13))
    local function refresh() val.Text = fmt(get()) end
    refresh()
    local function bump(d)
        PlaySFX("Click")
        set(math.clamp(get() + d, min, max))
        refresh()
        SaveConfig()
    end
    Track(minus.MouseButton1Click:Connect(function() bump(-step) end))
    Track(plus.MouseButton1Click:Connect(function() bump(step) end))
    return card
end

local function Segmented(parent, y, labels, values, current, onPick)
    local n = #labels
    local w = (0.94 - 0.02 * (n - 1)) / n
    local btns, sel = {}, current
    local function paint()
        for i, b in ipairs(btns) do StyleState(b, values[i] == sel) end
    end
    for i = 1, n do
        local b = New("TextButton", {
            Size = UDim2.new(w, 0, 0, 34), Position = UDim2.new(0.03 + (i - 1) * (w + 0.02), 0, 0, y),
            Text = labels[i], Font = Enum.Font.GothamBold, TextSize = 11, AutoButtonColor = false,
        }, parent)
        Round(b, 8)
        btns[i] = b
        Track(b.MouseButton1Click:Connect(function()
            PlaySFX("Click")
            sel = values[i]
            paint()
            onPick(values[i])
            SaveConfig()
        end))
    end
    OnTheme(paint)
end

local function MakeSlider(parent, y, min, max, init, onChange, onDone)
    local bg = New("Frame", {Size = UDim2.new(1, -28, 0, 10), Position = UDim2.fromOffset(14, y)}, parent)
    Bind(bg, "BackgroundColor3", "ButtonSolid"); Round(bg, 8)
    local fill = New("Frame", {Size = UDim2.new(math.clamp((init - min) / (max - min), 0, 1), 0, 1, 0)}, bg)
    Bind(fill, "BackgroundColor3", "Accent"); Round(fill, 8)
    local hit = New("TextButton", {Size = UDim2.new(1, 0, 0, 28), Position = UDim2.new(0, 0, 0.5, -14), BackgroundTransparency = 1, Text = ""}, bg)
    local dragging = false
    local function setFromX(x)
        local r = math.clamp((x - bg.AbsolutePosition.X) / math.max(1, bg.AbsoluteSize.X), 0, 1)
        fill.Size = UDim2.new(r, 0, 1, 0)
        onChange(min + r * (max - min))
    end
    Track(hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(i.Position.X)
        end
    end))
    Track(UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            setFromX(i.Position.X)
        end
    end))
    Track(UserInputService.InputEnded:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
            dragging = false
            if onDone then onDone() end
        end
    end))
end

local Rejoining, RejoinAttempts = false, 0

local function Rejoin(reason)
    if Rejoining then return end
    Rejoining = true
    Notify("Rejoining...", reason)
    SendWebhook("Rejoining", reason)
    SaveConfig()
    task.spawn(function()
        task.wait(1.5)
        local ok = pcall(function()
            if #Players:GetPlayers() <= 1 then
                TeleportService:Teleport(game.PlaceId, Player)
            else
                TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, Player)
            end
        end)
        if not ok then pcall(function() TeleportService:Teleport(game.PlaceId, Player) end) end
        task.wait(30)
        Rejoining = false
    end)
end

local function HopServer(reason)
    if Rejoining then return end
    Rejoining = true
    Notify("Server Hop", reason)
    SendWebhook("Server Hop", reason)
    pcall(function() TeleportService:Teleport(game.PlaceId, Player) end)
    task.delay(30, function() Rejoining = false end)
end

Track(TeleportService.TeleportInitFailed:Connect(function()
    RejoinAttempts += 1
    if RejoinAttempts <= 5 then
        task.wait(5 * RejoinAttempts)
        pcall(function() TeleportService:Teleport(game.PlaceId, Player) end)
    end
end))

local Potato = {On = false, Token = 0, Saved = setmetatable({}, {__mode = "k"}), Conns = {}, Env = nil}
local FX_CLASSES = {ParticleEmitter = true, Trail = true, Beam = true, Fire = true, Smoke = true, Sparkles = true}

local function PotatoTouch(o)
    if o:IsA("Terrain") then return end
    if o:IsA("BasePart") then
        if not Potato.Saved[o] then
            Potato.Saved[o] = {Material = o.Material, Reflectance = o.Reflectance, CastShadow = o.CastShadow}
        end
        o.Material = Enum.Material.SmoothPlastic
        o.Reflectance = 0
        o.CastShadow = false
    elseif FX_CLASSES[o.ClassName] or o:IsA("PostEffect") then
        if not Potato.Saved[o] then Potato.Saved[o] = {Enabled = o.Enabled} end
        o.Enabled = false
    end
end

local function PotatoSet(on, silent)
    Potato.Token += 1
    local token = Potato.Token
    if on then
        Potato.On = true
        if not Potato.Env then
            local E = {}
            pcall(function() E.GlobalShadows = Lighting.GlobalShadows end)
            pcall(function() E.Diffuse = Lighting.EnvironmentDiffuseScale end)
            pcall(function() E.Specular = Lighting.EnvironmentSpecularScale end)
            pcall(function()
                local T = workspace.Terrain
                E.Deco, E.WaveSize, E.WaveSpeed, E.WaterRefl = T.Decoration, T.WaterWaveSize, T.WaterWaveSpeed, T.WaterReflectance
            end)
            pcall(function() E.Quality = settings().Rendering.QualityLevel end)
            pcall(function()
                E.Volume = UserSettings():GetService("UserGameSettings").MasterVolume
            end)
            Potato.Env = E
        end
        pcall(function() Lighting.GlobalShadows = false end)
        pcall(function() Lighting.EnvironmentDiffuseScale = 0 end)
        pcall(function() Lighting.EnvironmentSpecularScale = 0 end)
        pcall(function()
            local T = workspace.Terrain
            T.Decoration = false; T.WaterWaveSize = 0; T.WaterWaveSpeed = 0; T.WaterReflectance = 0
        end)
        pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
        pcall(function() UserSettings():GetService("UserGameSettings").MasterVolume = 0 end)

        for _, root in ipairs({workspace, Lighting}) do
            table.insert(Potato.Conns, root.DescendantAdded:Connect(function(o)
                if Potato.On then pcall(PotatoTouch, o) end
            end))
        end
        task.spawn(function()
            for _, root in ipairs({workspace, Lighting}) do
                local list = root:GetDescendants()
                for i, o in ipairs(list) do
                    if Potato.Token ~= token or not Alive then return end
                    pcall(PotatoTouch, o)
                    if i % 300 == 0 then task.wait() end
                end
            end
            if not silent then Notify("Potato Mode", "Graphics reduced. Original values are saved") end
        end)
    else
        Potato.On = false
        for _, c in ipairs(Potato.Conns) do pcall(function() c:Disconnect() end) end
        table.clear(Potato.Conns)
        task.spawn(function()
            local n = 0
            for o, props in pairs(Potato.Saved) do
                if Potato.Token ~= token then return end
                if o.Parent then
                    for k, v in pairs(props) do pcall(function() o[k] = v end) end
                end
                Potato.Saved[o] = nil
                n += 1
                if n % 300 == 0 then task.wait() end
            end
            local E = Potato.Env
            if E then
                pcall(function() if E.GlobalShadows ~= nil then Lighting.GlobalShadows = E.GlobalShadows end end)
                pcall(function() if E.Diffuse then Lighting.EnvironmentDiffuseScale = E.Diffuse end end)
                pcall(function() if E.Specular then Lighting.EnvironmentSpecularScale = E.Specular end end)
                pcall(function()
                    local T = workspace.Terrain
                    if E.Deco ~= nil then T.Decoration = E.Deco end
                    if E.WaveSize then T.WaterWaveSize = E.WaveSize end
                    if E.WaveSpeed then T.WaterWaveSpeed = E.WaveSpeed end
                    if E.WaterRefl then T.WaterReflectance = E.WaterRefl end
                end)
                pcall(function() if E.Quality then settings().Rendering.QualityLevel = E.Quality end end)
                pcall(function() if E.Volume then UserSettings():GetService("UserGameSettings").MasterVolume = E.Volume end end)
                Potato.Env = nil
            end
            if not silent and Alive then Notify("Potato Mode", "Graphics restored") end
        end)
    end
end

                local hum = Player.Character and Player.Character:FindFirstChildOfClass("Humanoid")
                if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
            end
        end)
    end)
end

-- FPS / ping watcher (Heartbeat keeps running when RenderStepped throttles)
local frames, lastStat, lastLagAlert = 0, tick(), 0
Track(RunService.Heartbeat:Connect(function()
    frames += 1
    local now = tick()
    if now - lastStat < 1 then return end
    local fps = math.floor(frames / (now - lastStat))
    frames, lastStat = 0, now

    local ping = 0
    pcall(function() ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) end)

    if now - Session.StartTime > 10 then
        if fps < Session.LowestFps then Session.LowestFps = fps end
        if ping > Session.PeakPing then Session.PeakPing = ping end
        if ping >= Config.LagPingThreshold and now - lastLagAlert > 120 then
            lastLagAlert = now
            Notify("⚠️ Lag Spike", "Ping hit " .. ping .. "ms")
            if Config.WebhookLagAlerts then SendWebhook("Lag Spike", "Ping reached **" .. ping .. "ms**") end
        end
    end

    PerfLabel.Text = string.format("FPS: %d  •  Ping: %dms  •  %s", fps, ping, Config.Keybind.Name)
    HealthValue.Text = string.format("Now: %d FPS / %dms\nPeak ping: %dms  •  Lowest FPS: %d",
        fps, ping, Session.PeakPing, Session.LowestFps == 999 and fps or Session.LowestFps)
end))

-- Main 1-second loop
local lastPetTick, lastPetSave = tick(), tick()
task.spawn(function()
    while Alive and Gui.Parent do
        task.wait(1)
        local now = tick()
        local active = ActiveSeconds()

        if Config.Enabled then
            if now - Session.LastPulse >= Config.Interval then ExecuteAntiAFKPulse() end

            if Config.AfkAlarmHours > 0 and not Session.AlarmFired and active >= Config.AfkAlarmHours * 3600 then
                Session.AlarmFired = true
                PlaySFX("Alarm")
                Notify("AFK Milestone", "Active for " .. Config.AfkAlarmHours .. " hour(s)!")
                SendWebhook("AFK Milestone", "Active for **" .. Config.AfkAlarmHours .. "** hour(s).")
            end

            if Config.WebhookDigest and now - Session.LastHourlyDigest >= 3600 then
                Session.LastHourlyDigest = now
                SendWebhook("📊 Hourly AFK Digest", string.format("• **Time active:** %s\n• **Pulses:** %d\n• **Peak ping:** %dms\n• **Lowest FPS:** %d",
                    FormatDuration(active), Session.PulsesBlocked, Session.PeakPing, Session.LowestFps == 999 and 0 or Session.LowestFps))
            end

            local timeStr = string.format("%02d:%02d:%02d", math.floor(active / 3600), math.floor((active % 3600) / 60), active % 60)
            TimerValue.Text = timeStr
            TWLabel.Text = "🐾 " .. timeStr
            NextValue.Text = FormatDuration(math.max(0, Config.Interval - (now - Session.LastPulse)))
        else
            NextValue.Text = "Paused"
        end
        PulseValue.Text = tostring(Session.PulsesBlocked)

        if Config.ScheduledRejoinHours > 0 and now - Session.StartTime >= Config.ScheduledRejoinHours * 3600 then
            Rejoin("Scheduled rejoin (" .. Config.ScheduledRejoinHours .. "h)")
        end

        -- pet happiness decays slowly; saved every 5 minutes
        if now - lastPetTick >= 60 then
            lastPetTick = now
            Config.PetHappiness = math.max(0, Config.PetHappiness - 1)
            RefreshPet()
        end
        if now - lastPetSave >= 300 then
            lastPetSave = now
            SaveConfig()
        end
    end
end)
