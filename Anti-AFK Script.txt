local Bound = setmetatable({}, {__mode = "k"})

local function Dur(t)
	if not Config.Animations then return 0 end
	return t / math.max(0.25, Config.AnimSpeed)
end

local function Tween(obj, props, t, style, dir)
	local d = Dur(t or 0.2)
	if d <= 0 then
		for k, v in pairs(props) do pcall(function() obj[k] = v end) end
		return nil
	end
	local tw = TweenService:Create(obj, TweenInfo.new(d, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)
	tw:Play()
	return tw
end

local function Bind(obj, prop, key)
	Bound[obj] = Bound[obj] or {}
	Bound[obj][prop] = key
	obj[prop] = Theme[key]
	return obj
end
local function SetKey(obj, prop, key, t)
	Bound[obj] = Bound[obj] or {}
	Bound[obj][prop] = key
	Tween(obj, {[prop] = Theme[key]}, t or 0.2)
end

local function New(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props or {}) do o[k] = v end
	if parent then o.Parent = parent end
	return o
end
local function Round(o, r) return New("UICorner", {CornerRadius = UDim.new(0, r)}, o) end
local function Outline(o, key, th, tr)
	local s = New("UIStroke", {Thickness = th or 1, Transparency = tr or 0, ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, o)
	Bind(s, "Color", key)

end
local function Lbl(parent, text, size, key, font, pos, sz, xa)
	local l = New("TextLabel", {
		BackgroundTransparency = 1, Text = text, TextSize = size or 12, Font = font or Enum.Font.GothamMedium,

		TextXAlignment = xa or Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,
	}, parent)
	Bind(l, "TextColor3", key or "Text")
	return l
end

-- ESP -----------------------------------------------------------------
local function espClear(p)
	local o = FX.esp.objs[p]
	if o then
		pcall(function() o.hl:Destroy() end)
		pcall(function() o.bb:Destroy() end)
		FX.esp.objs[p] = nil
	end
end
local function espClearAll() for p in pairs(FX.esp.objs) do espClear(p) end end
local function espUpdate(S, dt)
	local esp = FX.esp
	esp.t += dt
	if esp.t < 0.1 then return end
	esp.t = 0
	local hue = S.rainbow and (tick() * 0.25) % 1 or S.fillHue
	local fill = Color3.fromHSV(hue, 0.85, 1)
	local outline = S.whiteOutline and Color3.new(1, 1, 1) or fill
	local myRoot = root()
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= player then
			local ch = p.Character
			local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
			local hu = ch and ch:FindFirstChildOfClass("Humanoid")
			local skip = S.teamCheck and p.Team ~= nil and p.Team == player.Team
			local dist = (hrp and myRoot) and (hrp.Position - myRoot.Position).Magnitude or 0
			if not ch or not hrp or skip or dist > S.maxDist then
				espClear(p)
			else
				local o = esp.objs[p]
				if not o or o.char ~= ch or not o.hl.Parent then
					espClear(p)
					local hl = New("Highlight", {Name = "UH_ESP", Adornee = ch, DepthMode = Enum.HighlightDepthMode.AlwaysOnTop}, ch)
					local bb = New("BillboardGui", {Name = "UH_Tag", Adornee = hrp, Size = UDim2.fromOffset(170, 52), StudsOffset = Vector3.new(0, 3.6, 0), AlwaysOnTop = true}, hrp)
					local l = New("TextLabel", {Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Font = Enum.Font.GothamBold, TextSize = 13, TextStrokeTransparency = 0.4}, bb)
					o = {hl = hl, bb = bb, lbl = l, char = ch}
					esp.objs[p] = o
				end
				o.hl.FillColor = fill
				o.hl.OutlineColor = outline
				o.hl.FillTransparency = S.fillTrans
				local lines = {}
				if S.showNames then lines[#lines + 1] = p.DisplayName end
				if S.showDist and myRoot then lines[#lines + 1] = math.floor(dist) .. " studs" end
				if S.showHealth and hu then lines[#lines + 1] = "HP " .. math.floor(hu.Health) .. "/" .. math.floor(hu.MaxHealth) end
				o.bb.Enabled = #lines > 0
				o.lbl.Text = table.concat(lines, "\n")
				o.lbl.TextColor3 = fill
			end
		end
	end
end

-- Tracers -------------------------------------------------------------
local function tracerClear()
	for p, l in pairs(FX.tr.lines) do pcall(function() l:Destroy() end) FX.tr.lines[p] = nil end
end
local function tracerUpdate(S)
	local c = cam()
	if not c or not Overlay then return end
	local vp = c.ViewportSize
	local origin = (S.origin == 0 and Vector2.new(vp.X / 2, 0)) or (S.origin == 1 and Vector2.new(vp.X / 2, vp.Y / 2)) or Vector2.new(vp.X / 2, vp.Y)
	local hue = S.rainbow and (tick() * 0.25) % 1 or S.hue
	local col = Color3.fromHSV(hue, 0.85, 1)
	local seen = {}
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= player then
			local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
			local skip = S.teamCheck and p.Team ~= nil and p.Team == player.Team
			if hrp and not skip then
				local v = c:WorldToViewportPoint(hrp.Position)
				if v.Z > 0 then
					local line = FX.tr.lines[p]
					if not line or not line.Parent then
						line = New("Frame", {Name = "UH_Tracer", AnchorPoint = Vector2.new(0.5, 0.5), BorderSizePixel = 0, BackgroundTransparency = 0.1}, Overlay)
						FX.tr.lines[p] = line
					end
					local target = Vector2.new(v.X, v.Y)
					local d = target - origin
					line.BackgroundColor3 = col
					line.Size = UDim2.fromOffset(d.Magnitude, S.thickness)
					line.Position = UDim2.fromOffset((origin.X + target.X) / 2, (origin.Y + target.Y) / 2)
					line.Rotation = math.deg(math.atan2(d.Y, d.X))
					seen[p] = true
				end
			end
		end
	end
	for p, l in pairs(FX.tr.lines) do
		if not seen[p] then l:Destroy() FX.tr.lines[p] = nil end
	end
end

-- Crosshair -----------------------------------------------------------
local function crossBuild(S)
	local cr = FX.cross
	if cr.root then cr.root:Destroy() cr.root = nil end
	if not Overlay then return end
	local col = Color3.fromHSV(S.hue, 0.85, 1)
	local rt = New("Frame", {Name = "UH_Cross", AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(0, 0), BackgroundTransparency = 1}, Overlay)
	local function bar(x, y, w, h)
		local f = New("Frame", {Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h), BackgroundColor3 = col, BorderSizePixel = 0}, rt)
		New("UIStroke", {Thickness = 1, Color = Color3.new(0, 0, 0), ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, f)
	end
	local s, g, t = S.size, S.gap, S.thick
	bar(-t / 2, -(g + s), t, s)
	bar(-t / 2, g, t, s)
	bar(-(g + s), -t / 2, s, t)
	bar(g, -t / 2, s, t)
	if S.dot then bar(-1.5, -1.5, 3, 3) end
	cr.root = rt
end

-- Fly -----------------------------------------------------------------
local function flyStop()
	local fly = FX.fly
	if fly.bv then fly.bv:Destroy() fly.bv = nil end
	if fly.bg then fly.bg:Destroy() fly.bg = nil end
	local h = hum()
	if h then h.PlatformStand = false end
end

-- X-Ray ---------------------------------------------------------------
local function xrApply(on, S)
	local xr = FX.xr
	xr.token = (xr.token or 0) + 1
	local tk = xr.token
	if not on then
		for p in pairs(xr.parts) do
			if p.Parent then p.LocalTransparencyModifier = 0 end
			xr.parts[p] = nil
		end
		return
	end
	task.spawn(function()
		local n = 0
		for _, d in ipairs(workspace:GetDescendants()) do
			if xr.token ~= tk then return end
			if d:IsA("BasePart") and not d:IsA("Terrain") then
				local m = d:FindFirstAncestorOfClass("Model")
				if not (m and m:FindFirstChildOfClass("Humanoid")) then
					d.LocalTransparencyModifier = S.amount
					xr.parts[d] = true
				end
			end
			n += 1
			if n % 1500 == 0 then task.wait() end
		end
	end)
end

-- Dragging
local function MakeDraggable(obj, handle)
	local dragging, dragStart, startPos
	Track(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging, dragStart, startPos = true, input.Position, obj.Position
			local c
			c = input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false c:Disconnect() end
			end)
		end
	end))
	Track(UIS.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			obj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end))
end

local function Ripple(parent, x, y)
	if not Config.Animations then return end
	local s = math.max(0.1, UIScaleObj.Scale)
	local rel = (Vector2.new(x, y) - parent.AbsolutePosition) / s
	local d = math.max(parent.AbsoluteSize.X, parent.AbsoluteSize.Y) / s * 2.2
	local c = New("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromOffset(rel.X, rel.Y), Size = UDim2.fromOffset(0, 0), BackgroundColor3 = Color3.new(1, 1, 1), BackgroundTransparency = 0.82, BorderSizePixel = 0, ZIndex = 0}, parent)
	New("UICorner", {CornerRadius = UDim.new(1, 0)}, c)
	Tween(c, {Size = UDim2.fromOffset(d, d), BackgroundTransparency = 1}, 0.55, Enum.EasingStyle.Quad)
	task.delay(Dur(0.55) + 0.05, function() c:Destroy() end)
end

local function Btn(parent, text, pos, size, style, onClick)
	local b = New("TextButton", {Text = text, Position = pos, Size = size, Font = Enum.Font.GothamBold, TextSize = 12, AutoButtonColor = false, ClipsDescendants = true, BorderSizePixel = 0}, parent)
	Round(b, 8)
	local bg, hv, tx = "Switch", "SwitchHover", "Text"
	if style == "accent" then bg, hv, tx = "Accent", "Accent2", "OnAccent"
	elseif style == "danger" then bg, hv, tx = "Danger", "DangerHover", "White" end
	Bind(b, "BackgroundColor3", bg); Bind(b, "TextColor3", tx)
	Track(b.MouseEnter:Connect(function() SetKey(b, "BackgroundColor3", hv, 0.15) end))
	Track(b.MouseLeave:Connect(function() SetKey(b, "BackgroundColor3", bg, 0.15) end))
	Track(b.MouseButton1Click:Connect(function()
		local m = UIS:GetMouseLocation()
		Ripple(b, m.X, m.Y)
		if onClick then onClick(b) end
	end))
	return b
end

local function MakeSwitch(parent, pos, w, h, state, clickable)
	local sw = New("Frame", {Position = pos, Size = UDim2.fromOffset(w, h), BorderSizePixel = 0}, parent)
	Round(sw, h / 2)
	local pad = 3
	local kd = h - pad * 2
	local knob = New("Frame", {Size = UDim2.fromOffset(kd, kd), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0}, sw)
	Round(knob, kd / 2)
	local api = {frame = sw}
	if clickable then api.button = New("TextButton", {Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = ""}, sw) end
	function api.set(on, instant)
		local kp = on and UDim2.fromOffset(w - kd - pad, pad) or UDim2.fromOffset(pad, pad)
		local key = on and "Accent" or "Switch"
		if instant then
			knob.Position = kp
			SetKey(sw, "BackgroundColor3", key, 0)
		else
			Tween(knob, {Position = kp}, 0.3, Enum.EasingStyle.Back)
			SetKey(sw, "BackgroundColor3", key, 0.25)
		end
	end
	api.set(state, true)
	return api
end

local SET_H = 42
-- Builds one setting control (slider / hue / check). Returns {set = fn}
local function BuildSetting(parent, y, def, store, onChange)
	local api = {}
	if def.type == "check" then
		Lbl(parent, def.name, 12, "Text", Enum.Font.GothamMedium, UDim2.fromOffset(14, y + 8), UDim2.new(1, -80, 0, 18))
		local val = store[def.id]
		local sw = MakeSwitch(parent, UDim2.new(1, -56, 0, y + 7), 38, 20, val, true)
		Track(sw.button.MouseButton1Click:Connect(function()
			val = not val
			store[def.id] = val
			sw.set(val)
			onChange(def.id, val)
			SaveConfig()
		end))
		function api.set(v) val = v store[def.id] = v sw.set(v, true) end
		return api
	end

	Lbl(parent, def.name, 12, "Text", Enum.Font.GothamMedium, UDim2.fromOffset(14, y), UDim2.new(0.6, 0, 0, 16))
	local valLbl = Lbl(parent, "", 11, "Muted", Enum.Font.GothamBold, UDim2.new(0.5, 0, 0, y), UDim2.new(0.5, -14, 0, 16), Enum.TextXAlignment.Right)
	local bar = New("Frame", {Size = UDim2.new(1, -28, 0, 6), Position = UDim2.fromOffset(14, y + 26), BorderSizePixel = 0, BackgroundColor3 = Color3.new(1, 1, 1)}, parent)
	Round(bar, 3)
	local fill
	if def.type == "hue" then
		New("UIGradient", {Color = RAINBOW}, bar)
	else
		Bind(bar, "BackgroundColor3", "Switch")
		fill = New("Frame", {Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0}, bar)
		Bind(fill, "BackgroundColor3", "Accent"); Round(fill, 3)
	end
	local knob = New("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(14, 14), BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Position = UDim2.new(0, 0, 0.5, 0)}, bar)
	Round(knob, 7)
	local ks = New("UIStroke", {Thickness = 2}, knob)
	Bind(ks, "Color", "Accent")
	local hit = New("TextButton", {Size = UDim2.new(1, 0, 0, 28), Position = UDim2.new(0, 0, 0.5, -14), BackgroundTransparency = 1, Text = ""}, bar)

	local fmt = def.fmt or function(v)
		if def.type == "hue" then return math.floor(v * 360) .. "°" end
		if (def.step or 1) >= 1 then return tostring(math.floor(v + 0.5)) end
		return string.format("%.2f", v)
	end

	local function setValue(v, silent, animate)
		v = math.clamp(v, def.min or 0, def.max or 1)
		local st = def.step or (def.type == "hue" and 0.01) or 1
		v = math.floor(v / st + 0.5) * st
		v = math.floor(v * 10000 + 0.5) / 10000
		store[def.id] = v
		local r = (v - (def.min or 0)) / ((def.max or 1) - (def.min or 0))
		if animate then
			Tween(knob, {Position = UDim2.new(r, 0, 0.5, 0)}, 0.25)
			if fill then Tween(fill, {Size = UDim2.new(r, 0, 1, 0)}, 0.25) end
		else
			knob.Position = UDim2.new(r, 0, 0.5, 0)
			if fill then fill.Size = UDim2.new(r, 0, 1, 0) end
		end
		if def.type == "hue" then knob.BackgroundColor3 = Color3.fromHSV(v, 0.85, 1) end
		valLbl.Text = fmt(v)
		if not silent then onChange(def.id, v) end
	end

	Track(hit.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			local function fromX(x)
				local r = math.clamp((x - bar.AbsolutePosition.X) / math.max(1, bar.AbsoluteSize.X), 0, 1)
				setValue((def.min or 0) + r * ((def.max or 1) - (def.min or 0)))
			end
			Tween(knob, {Size = UDim2.fromOffset(18, 18)}, 0.15, Enum.EasingStyle.Back)
			ActiveDrag = {move = fromX, done = function() Tween(knob, {Size = UDim2.fromOffset(14, 14)}, 0.2) end}
			fromX(i.Position.X)
		end
	end))

	setValue(store[def.id], true, false)
	function api.set(v, animate) setValue(v, true, animate) end
	return api
end

-- Segmented picker (wraps into several rows when perRow < #options)
local function MakeSegmented(parent, y, options, current, onPick, perRow)
	local n = #options
	perRow = perRow or n
	local gap = 6
	local rows = math.ceil(n / perRow)
	local btns = {}
	local sel = current
	local function paint(animate)
		for _, o in ipairs(btns) do
			local on = (o.value == sel)
			SetKey(o.b, "BackgroundColor3", on and "Accent" or "Switch", animate and 0.2 or 0)
			SetKey(o.b, "TextColor3", on and "OnAccent" or "Muted", animate and 0.2 or 0)
		end
	end
	for i, opt in ipairs(options) do
		local col, rw = (i - 1) % perRow, math.floor((i - 1) / perRow)
		local b = New("TextButton", {
			Size = UDim2.new(1 / perRow, -gap * (perRow - 1) / perRow, 0, 30),
			Position = UDim2.new(col / perRow, col * gap / perRow, 0, y + rw * 36),
			Text = opt.label, Font = Enum.Font.GothamBold, TextSize = 11, AutoButtonColor = false, BorderSizePixel = 0,
		}, parent)
		Round(b, 8)
		btns[i] = {b = b, value = opt.value}
		Track(b.MouseButton1Click:Connect(function()
			sel = opt.value
			paint(true)
			onPick(opt.value)
			SaveConfig()
		end))
	end
	paint(false)
	return {set = function(v) sel = v paint(true) end}, 30 + (rows - 1) * 36
end

local KeyMap = {LeftControl = "LCtrl", RightControl = "RCtrl", LeftShift = "LShift", RightShift = "RShift", LeftAlt = "LAlt", RightAlt = "RAlt", Backspace = "Back", Return = "Enter"}
local function KeyLabel(name)
	if name == nil or name == "" then return "NONE" end
	return string.upper(KeyMap[name] or name):sub(1, 7)
end

local function SetFeature(id, on, silent)
	local f, st = FeatureById[id], State[id]
	if not f or st.on == on then return end
	st.on = on
	for _, v in ipairs(st.visuals) do pcall(v, on) end
	if f.apply then
		local ok, err = pcall(f.apply, on, Config.Features[id].s)
		if not ok then warn("[UtilityHub] " .. id .. ": " .. tostring(err)) end
	end
	RefreshBadges()
	if id == "fpscap" and on and not setfpscap then Notify("FPS Cap", "Your executor has no setfpscap", "warn", true) end
	if not silent then Notify(f.name, on and "Enabled" or "Disabled", on and "success" or "info") end
end

	Track(RunService.Heartbeat:Connect(function(dt)
		local mode = Troll.mode
		if not mode then return end
		local r = root()
		if not r then return end
		local S = Troll.settings

		-- while a troll mode runs, our own character never collides with anything (no shoving anyone)
		local ch = player.Character
		if ch then
			for _, p in ipairs(ch:GetDescendants()) do
				if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
			end
		end

		if mode == "Spin" then
			r.CFrame *= CFrame.Angles(0, dt * S.speed * 2, 0)
			r.AssemblyLinearVelocity = Vector3.zero
			return
		end

		local t = Troll.target
		local tr = t and t.Character and t.Character:FindFirstChild("HumanoidRootPart")
		if not tr then return end

		-- fast phase (speed slider) and slow phase (breathing / height sweeps)
		Troll.angle = (Troll.angle + dt * S.speed * 2) % (math.pi * 2)
		Troll.phase = (Troll.phase + dt * 1.2) % (math.pi * 2)
		local a, ph = Troll.angle, Troll.phase
		local R = math.max(S.radius, 4)
		local Hh = S.height
		local c = tr.Position
		local base = tr.CFrame
		local function face(pos) return CFrame.lookAt(pos, Vector3.new(c.X, pos.Y, c.Z)) end

		if mode == "Orbit" then
			r.CFrame = face(c + Vector3.new(math.cos(a) * R, 0, math.sin(a) * R))
		elseif mode == "Orbit Rev" then
			r.CFrame = face(c + Vector3.new(math.cos(a) * R, 0, -math.sin(a) * R))
		elseif mode == "Spiral" then
			local rr = math.max(4, R * (0.6 + 0.4 * math.sin(ph)))
			r.CFrame = face(c + Vector3.new(math.cos(a) * rr, 0, math.sin(a) * rr))
		elseif mode == "Helix" then
			local pos = c + Vector3.new(math.cos(a) * R, math.sin(ph * 1.5) * Hh, math.sin(a) * R)
			r.CFrame = CFrame.lookAt(pos, c)
		elseif mode == "Ellipse" then
			r.CFrame = face(c + Vector3.new(math.cos(a) * R * 1.8, 0, math.sin(a) * R))
		elseif mode == "Satellite" then
			local pos = c + Vector3.new(math.cos(a) * R, math.sin(a) * R * math.sin(0.8), math.sin(a) * R * math.cos(0.8))
			r.CFrame = CFrame.lookAt(pos, c)
		elseif mode == "Tornado" then
			local rr = R * (1 + 0.4 * math.cos(ph * 2))
			local pos = c + Vector3.new(math.cos(a) * rr, math.sin(ph * 2) * Hh, math.sin(a) * rr)
			r.CFrame = CFrame.new(pos) * CFrame.Angles(0, a * 1.7, 0)
		elseif mode == "Halo" then
			local rh = math.max(3.5, R * 0.5)
			r.CFrame = face(c + Vector3.new(math.cos(a) * rh, 6 + Hh * 0.25, math.sin(a) * rh))
		elseif mode == "Ferris" then
			local pos = (base * CFrame.new(0, math.sin(a) * R, math.cos(a) * R)).Position
			r.CFrame = CFrame.lookAt(pos, c)
		elseif mode == "Follow" then
			r.CFrame = base * CFrame.new(0, 0, R)
		elseif mode == "Stalker" then
			local goal = (base * CFrame.new(0, 0, R)).Position
			r.CFrame = r.CFrame:Lerp(face(goal), math.min(1, dt * 6))
		elseif mode == "Sidekick" then
			r.CFrame = base * CFrame.new(math.max(4, R * 0.5), 0, 0)
		elseif mode == "Face-off" then
			r.CFrame = face((base * CFrame.new(0, 0, -R)).Position)
		elseif mode == "Zigzag" then
			r.CFrame = face((base * CFrame.new(math.sin(a) * R, 0, R)).Position)
		elseif mode == "Pendulum" then
			local ang = math.sin(a) * 1.3
			r.CFrame = face(c + base:VectorToWorldSpace(Vector3.new(math.sin(ang) * R, 0, -math.cos(ang) * R)))
		elseif mode == "Bounce" then
			r.CFrame = face((base * CFrame.new(0, math.abs(math.sin(a)) * (Hh + 2), -R)).Position)
		elseif mode == "Blink" then
			Troll.blinkT = (Troll.blinkT or 0) + dt
			if Troll.blinkT >= math.max(0.03, 1 / (S.speed * 0.2)) or not Troll.blinkA then
				Troll.blinkT = 0
				Troll.blinkA = math.random() * math.pi * 2
			end
			r.CFrame = face(c + Vector3.new(math.cos(Troll.blinkA) * R, 0, math.sin(Troll.blinkA) * R))
		elseif mode == "Copycat" then
			local th, h = t.Character:FindFirstChildOfClass("Humanoid"), hum()
			if th and h then
				h:Move(th.MoveDirection, false)
				if th.Jump then h.Jump = true end
			end
			return
		elseif mode == "Headsit" then
			local h = hum()
			if h and not h.Sit then h.Sit = true end
			r.CFrame = base * CFrame.new(0, 2.6, 0)
		elseif mode == "Backpack" then
			r.CFrame = base * CFrame.new(0, 0, 1.2)
		elseif mode == "Shoulder" then
			r.CFrame = base * CFrame.new(1.8, 2.2, 0)
		elseif mode == "Hover" then
			r.CFrame = base * CFrame.new(0, 3 + R * 0.5, 0)
		elseif mode == "Stare" then
			r.CFrame = CFrame.lookAt(r.Position, Vector3.new(c.X, r.Position.Y, c.Z))
		end
		r.AssemblyLinearVelocity = Vector3.zero
	end))

		InfoVals[key] = Lbl(InfoCard, "-", 12, "Text", Enum.Font.GothamBold, UDim2.new(0.4, 0, 0, 10 + (i - 1) * 22), UDim2.new(0.6, -16, 0, 18), Enum.TextXAlignment.Right)
	end

	local function ib3(col, text, style, fn)

	ib3(0, "Teleport", "accent", function()

	ib3(1, "Spectate", "ghost", function()

		local order = 0
		for _, p in ipairs(list) do
			if q == "" or string.find(string.lower(p.DisplayName), q, 1, true) or string.find(string.lower(p.Name), q, 1, true) then
				order += 1

				Bind(av, "BackgroundColor3", "Accent"); Bind(av, "TextColor3", "Accent"); New("UICorner", {CornerRadius = UDim.new(1, 0)}, av)

local switchToken = 0
SwitchTab = function(name, instant)
	if CurTab == name or not Pages[name] then return end
	local old = CurTab
	CurTab = name
	Config.Tab = name
	SaveConfig()
	switchToken += 1
	local token = switchToken

	local tb = TabButtons[name]
	Tween(Pill, {Position = UDim2.fromOffset(10, TABY0 + (tb.index - 1) * 40)}, instant and 0 or 0.35, Enum.EasingStyle.Back)
	for n, t in pairs(TabButtons) do SetKey(t.label, "TextColor3", n == name and "Accent" or "Muted", 0.2) end
	TitleLbl.Text = name
	SubLbl.Text = TabSubs[name] or ""

	for n, p in pairs(Pages) do
		if n ~= name and n ~= old then p.group.Visible = false end
	end
	if old and Pages[old] then
		local og = Pages[old].group
		Tween(og, {GroupTransparency = 1}, 0.12)
		task.delay(Dur(0.12) + 0.02, function() if switchToken == token then og.Visible = false end end)
	end
	local pg = Pages[name].group
	pg.GroupTransparency = instant and 0 or 1
	pg.Position = UDim2.fromOffset(0, instant and 0 or 16)
	pg.Visible = true
	Pages[name].scroll.CanvasPosition = Vector2.zero
	if not instant then Tween(pg, {GroupTransparency = 0, Position = UDim2.fromOffset(0, 0)}, 0.4) end
end

Track(UIS.InputBegan:Connect(function(input, gpe)
	-- Click teleport
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		if gpe or not State.clicktp.on then return end
		local S = Config.Features.clicktp.s
		if S.ctrl and not (UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.RightControl)) then return end
		local c = player.Character
		if c and mouse.Hit then
			local cf = c:GetPivot()
			c:PivotTo(CFrame.new(mouse.Hit.Position + Vector3.new(0, 3, 0)) * (cf - cf.Position))
		end
		return
	end
	if input.UserInputType ~= Enum.UserInputType.Keyboard then return end

	if Listening then
		local l = Listening
		Listening = nil
		if input.KeyCode == Enum.KeyCode.Escape then l.onSet("")
		elseif input.KeyCode ~= Enum.KeyCode.Unknown then l.onSet(input.KeyCode.Name) end
		l.refresh()
		RefreshLauncher()
		return
	end
	if gpe or UIS:GetFocusedTextBox() then return end

	local name = input.KeyCode.Name
	if name == Config.MenuKey then SetOpen(not Visible) return end
	if Config.PanicKey ~= "" and name == Config.PanicKey then PanicAll() return end
	for _, f in ipairs(Features) do
		if Config.Features[f.id].key == name then SetFeature(f.id, not State[f.id].on) end
	end
end))

local jumpUsed, lastGroundJump = 0, 0
Track(UIS.JumpRequest:Connect(function()
	local h, r = hum(), root()
	if not h or not r then return end
	if State.infjump.on then
		h:ChangeState(Enum.HumanoidStateType.Jumping)
		return
	end
	if not State.multijump.on then return end
	local S = Config.Features.multijump.s
	if h.FloorMaterial ~= Enum.Material.Air then
		lastGroundJump, jumpUsed = tick(), 0
		return
	end
	if tick() - lastGroundJump > 0.18 and jumpUsed < S.extra then
		jumpUsed += 1
		lastGroundJump = tick()
		local v = r.AssemblyLinearVelocity
		r.AssemblyLinearVelocity = Vector3.new(v.X, S.power, v.Z)
	end
end))

Cleanup = function(animated)
	if not Alive then return end
	Alive = false
	WriteConfig()
	for _, f in ipairs(Features) do
		if State[f.id].on and f.apply then pcall(f.apply, false, Config.Features[f.id].s) end
	end
	espClearAll()
	tracerClear()
	pcall(StopSpectate)
	pcall(TrollStop)
	for _, c in ipairs(Connections) do pcall(function() c:Disconnect() end) end
	table.clear(Connections)
	env.UtilityHubCleanup = nil
	if animated and Main.Visible then
		Tween(Main, {GroupTransparency = 1}, 0.25)
		Tween(UIScaleObj, {Scale = Config.Scale * 0.9}, 0.25)
		task.delay(Dur(0.25) + 0.05, function() Gui:Destroy() end)
	else
		Gui:Destroy()
	end
end
env.UtilityHubCleanup = function() Cleanup(false) end

--====================================================================
--   SPLASH + STARTUP
--====================================================================
ApplyThemeVars()

task.spawn(function()
	local sp = New("CanvasGroup", {Size = UDim2.fromOffset(340, 160), Position = UDim2.new(0.5, -170, 0.5, -80), BorderSizePixel = 0, GroupTransparency = 1}, Gui)
	Bind(sp, "BackgroundColor3", "Bg"); Round(sp, 16)
	local sst = New("UIStroke", {Thickness = 1.5}, sp); Bind(sst, "Color", "Accent")
	local em = New("Frame", {Size = UDim2.fromOffset(46, 46), Position = UDim2.new(0.5, -23, 0, 20), BorderSizePixel = 0, BackgroundColor3 = Color3.new(1, 1, 1)}, sp)
	Round(em, 13)
	AccentGrads[#AccentGrads + 1] = New("UIGradient", {Rotation = 45, Color = ColorSequence.new(Theme.Accent, Theme.Accent2)}, em)
	New("TextLabel", {Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "UH", Font = Enum.Font.GothamBlack, TextSize = 18, TextColor3 = Color3.fromRGB(12, 12, 18)}, em)
	Lbl(sp, "UTILITY HUB", 18, "Text", Enum.Font.GothamBlack, UDim2.new(0, 0, 0, 74), UDim2.new(1, 0, 0, 24), Enum.TextXAlignment.Center)
	local status = Lbl(sp, "Starting…", 11, "Muted", Enum.Font.Gotham, UDim2.new(0, 0, 0, 100), UDim2.new(1, 0, 0, 16), Enum.TextXAlignment.Center)
	local track = New("Frame", {Size = UDim2.new(1, -60, 0, 4), Position = UDim2.fromOffset(30, 128), BorderSizePixel = 0}, sp)
	Bind(track, "BackgroundColor3", "Switch"); Round(track, 2)
	local fill = New("Frame", {Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0}, track)
	Bind(fill, "BackgroundColor3", "Accent"); Round(fill, 2)

	Tween(sp, {GroupTransparency = 0}, 0.4)
	for _, step in ipairs({{"Loading modules…", 0.3}, {"Reading config…", 0.6}, {"Building interface…", 0.9}, {"Ready", 1}}) do
		if not Alive then return end
		status.Text = step[1]
		Tween(fill, {Size = UDim2.new(step[2], 0, 1, 0)}, 0.35)
		task.wait(math.max(Dur(0.35), 0.04) + 0.1)
	end
	Tween(sp, {GroupTransparency = 1}, 0.3)
	task.wait(math.max(Dur(0.3), 0.04) + 0.05)
	sp:Destroy()
	if not Alive then return end

	SwitchTab(TabButtons[Config.Tab] and Config.Tab or "Home", true)
	SetOpen(true)
	Notify("Utility Hub v" .. VERSION, "Press " .. KeyLabel(Config.MenuKey) .. " to show or hide", "success")
end)
